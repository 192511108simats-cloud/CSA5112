# Caesar Cipher

def encrypt(text, key):
    cipher = ""

    for ch in text:

        if ch.isupper():
            cipher += chr((ord(ch) - ord('A') + key) % 26 + ord('A'))

        elif ch.islower():
            cipher += chr((ord(ch) - ord('a') + key) % 26 + ord('a'))

        else:
            cipher += ch

    return cipher


def decrypt(text, key):
    plain = ""

    for ch in text:

        if ch.isupper():
            plain += chr((ord(ch) - ord('A') - key) % 26 + ord('A'))

        elif ch.islower():
            plain += chr((ord(ch) - ord('a') - key) % 26 + ord('a'))

        else:
            plain += ch

    return plain


try:

    while True:

        plaintext = input("\nEnter Plain Text : ")

        while True:
            try:
                key = int(input("Enter Key        : "))
                break
            except ValueError:
                print("Please enter a valid integer key.")

        encrypted = encrypt(plaintext, key)
        decrypted = decrypt(encrypted, key)

        print("\nEncrypted Text :", encrypted)
        print("Decrypted Text :", decrypted)

except KeyboardInterrupt:
    print("\n\nProgram Closed.")
