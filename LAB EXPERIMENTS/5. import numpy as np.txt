import numpy as np

def mod_inverse(a, m):
    a = a % m
    for i in range(1, m):
        if (a * i) % m == 1:
            return i
    return None

def matrix_inverse(key):
    det = int(round(np.linalg.det(key)))
    det = det % 26

    det_inv = mod_inverse(det, 26)
    if det_inv is None:
        return None

    adj = np.array([[key[1][1], -key[0][1]],
                    [-key[1][0], key[0][0]]])

    inv = (det_inv * adj) % 26
    return inv

while True:
    try:
        text = input("Enter Plain Text : ").upper().replace(" ", "")

        if len(text) % 2 != 0:
            text += "X"

        print("\nEnter Key Matrix")

        a11 = int(input("Enter a11 : "))
        a12 = int(input("Enter a12 : "))
        a21 = int(input("Enter a21 : "))
        a22 = int(input("Enter a22 : "))

        key = np.array([[a11, a12],
                        [a21, a22]])

        print("\nKey Matrix")
        print(key)

        cipher = ""

        for i in range(0, len(text), 2):
            pair = np.array([[ord(text[i]) - 65],
                             [ord(text[i + 1]) - 65]])

            result = np.dot(key, pair) % 26

            cipher += chr(result[0][0] + 65)
            cipher += chr(result[1][0] + 65)

        print("\nEncrypted Text :", cipher)

        inv_key = matrix_inverse(key)

        if inv_key is None:
            print("Key Matrix is not invertible modulo 26.")
            continue

        plain = ""

        for i in range(0, len(cipher), 2):
            pair = np.array([[ord(cipher[i]) - 65],
                             [ord(cipher[i + 1]) - 65]])

            result = np.dot(inv_key, pair) % 26

            plain += chr(int(result[0][0]) + 65)
            plain += chr(int(result[1][0]) + 65)

        print("Decrypted Text :", plain)
        print()

    except KeyboardInterrupt:
        print("\nProgram Stopped.")
        break
